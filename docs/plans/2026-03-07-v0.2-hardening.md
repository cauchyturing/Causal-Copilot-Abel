```python
"""PC algorithm backend using causal-learn."""
import numpy as np
import pandas as pd
from typing import Any, Dict, Tuple

from ._base import Backend


class PCBackend(Backend):
    def fit(self, data: pd.DataFrame) -> Tuple[np.ndarray, Dict[str, Any], Any]:
        if 'domain_index' in data.columns:
            data = data.drop(columns=['domain_index'])

        from causallearn.search.ConstraintBased.PC import pc as pc_algo
        from causallearn.utils.cit import CIT

        cit = CIT(data.values, self._params.get('indep_test', 'fisherz'))
        result = pc_algo(
            data.values,
            alpha=self._params.get('alpha', 0.05),
            indep_test=cit,
            stable=self._params.get('stable', True),
            uc_rule=self._params.get('uc_rule', 0),
            uc_priority=self._params.get('uc_priority', 2),
            depth=self._params.get('depth', 4),
            show_progress=False,
        )
        adj_matrix = result.G.graph
        info = {'sepset': {}}
        return adj_matrix, info, result
```

```python
"""NOTEARS (Linear) backend using causal-learn."""
import numpy as np
import pandas as pd
from typing import Any, Dict, Tuple

from ._base import Backend


class NOTEARSLinearBackend(Backend):
    def fit(self, data: pd.DataFrame) -> Tuple[np.ndarray, Dict[str, Any], Any]:
        if 'domain_index' in data.columns:
            data = data.drop(columns=['domain_index'])

        from causallearn.search.ScoreBased.ExactSearch import bic_exact_search
        # NOTEARS uses the castle or causal-learn implementation
        try:
            from castle.algorithms import Notears
            model = Notears(
                lambda1=self._params.get('lambda1', 0.1),
                max_iter=self._params.get('max_iter', 100),
                h_tol=self._params.get('h_tol', 1e-8),
                w_threshold=self._params.get('w_threshold', 0.3),
            )
            model.learn(data.values)
            adj_matrix = model.causal_matrix.astype(int)
        except ImportError:
            # Fallback to causal-learn NOTEARS
            from causallearn.search.ScoreBased.NOTEARS import notears_linear
            W = notears_linear(
                data.values,
                lambda1=self._params.get('lambda1', 0.1),
                max_iter=self._params.get('max_iter', 100),
                h_tol=self._params.get('h_tol', 1e-8),
                w_threshold=self._params.get('w_threshold', 0.3),
            )
            adj_matrix = (np.abs(W) > 0).astype(int)
            model = None

        info = {}
        return adj_matrix, info, model
```

```python
"""PCMCI backend using tigramite."""
import numpy as np
import pandas as pd
from typing import Any, Dict, Tuple

from ._base import Backend


class PCMCIBackend(Backend):
    def fit(self, data: pd.DataFrame) -> Tuple[np.ndarray, Dict[str, Any], Any]:
        if 'domain_index' in data.columns:
            data = data.drop(columns=['domain_index'])

        from tigramite import data_processing as pp
        from tigramite.pcmci import PCMCI as TigramitePCMCI
        from tigramite.independence_tests.parcorr import ParCorr

        dataframe = pp.DataFrame(data.values, var_names=list(data.columns))
        parcorr = ParCorr(significance='analytic')
        pcmci = TigramitePCMCI(
            dataframe=dataframe,
            cond_ind_test=parcorr,
            verbosity=0,
        )
        results = pcmci.run_pcmci(
            tau_min=self._params.get('tau_min', 0),
            tau_max=self._params.get('tau_max', 2),
            pc_alpha=self._params.get('pc_alpha', 0.05),
            alpha_level=self._params.get('alpha_level', 0.05),
        )

        n_vars = data.shape[1]
        adj_matrix = np.zeros((n_vars, n_vars), dtype=int)
        p_matrix = results['p_matrix']
        val_matrix = results['val_matrix']
        alpha = self._params.get('alpha_level', 0.05)

        for i in range(n_vars):
            for j in range(n_vars):
                if i != j:
                    for tau in range(self._params.get('tau_max', 2) + 1):
                        if p_matrix[i, j, tau] < alpha:
                            adj_matrix[j, i] = 1
                            break

        info = {
            'p_matrix': p_matrix,
            'val_matrix': val_matrix,
        }
        return adj_matrix, info, pcmci
```

```python
"""Canonical algorithm registry -- single source of truth.

Every algorithm's metadata, adapter class, defaults, and upstream info
lives here. copilot.py, planner.py, cli.py, and docs all read from this.
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Dict, List, Type


@dataclass(frozen=True)
class AlgorithmSpec:
    name: str
    adapter_cls: Type                          # lazy — resolved at import
    family: str                                # constraint, score, functional, hybrid, timeseries
    default_params: Dict[str, Any]
    upstream_packages: List[str]               # e.g. ["causal-learn"]
    algorithm_version: str                     # e.g. "pc-v1|causallearn=0.1.3.8"
    tags: tuple = ()                           # e.g. ("dag-output", "linear")
    optional_deps: tuple = ()                  # pip extras needed


def _build_registry() -> Dict[str, AlgorithmSpec]:
    from causal_copilot.algorithms.adapters import (
        PCAdapter, GESAdapter, NOTEARSLinearAdapter,
        DirectLiNGAMAdapter, PCMCIAdapter,
    )

    return {
        "PC": AlgorithmSpec(
            name="PC",
            adapter_cls=PCAdapter,
            family="constraint",
            default_params={"alpha": 0.05, "indep_test": "fisherz", "stable": True, "depth": 4},
            upstream_packages=["causal-learn"],
            algorithm_version="pc-v1",
            tags=("cpdag-output", "linear", "gaussian"),
        ),
        "GES": AlgorithmSpec(
            name="GES",
            adapter_cls=GESAdapter,
            family="score",
            default_params={"score_func": "local_score_BIC"},
            upstream_packages=["causal-learn"],
            algorithm_version="ges-v1",
            tags=("cpdag-output", "linear"),
        ),
        "NOTEARSLinear": AlgorithmSpec(
            name="NOTEARSLinear",
            adapter_cls=NOTEARSLinearAdapter,
            family="score",
            default_params={"lambda1": 0.1, "max_iter": 100, "h_tol": 1e-8, "w_threshold": 0.3},
            upstream_packages=["castle", "causal-learn"],
            algorithm_version="notears-linear-v1",
            tags=("dag-output", "linear", "continuous-optimization"),
            optional_deps=("algorithms",),
        ),
        "DirectLiNGAM": AlgorithmSpec(
            name="DirectLiNGAM",
            adapter_cls=DirectLiNGAMAdapter,
            family="functional",
            default_params={"measure": "pwling"},
            upstream_packages=["causal-learn"],
            algorithm_version="direct-lingam-v1",
            tags=("dag-output", "linear", "non-gaussian"),
        ),
        "PCMCI": AlgorithmSpec(
            name="PCMCI",
            adapter_cls=PCMCIAdapter,
            family="timeseries",
            default_params={"tau_min": 0, "tau_max": 2, "pc_alpha": 0.05, "alpha_level": 0.05},
            upstream_packages=["tigramite"],
            algorithm_version="pcmci-v1",
            tags=("timeseries", "constraint"),
            optional_deps=("algorithms",),
        ),
    }


REGISTRY: Dict[str, AlgorithmSpec] = _build_registry()
```

```python
import multiprocessing as mp
import pickle

def _run_algorithm(algo, data, timeout):
    """Run algorithm in subprocess with timeout. Returns (adj, meta) or raises TimeoutError."""

    def _worker(conn, algo_cls, algo_params, data_bytes, columns):
        """Subprocess worker — import and run algorithm."""
        try:
            import pandas as pd
            import numpy as np
            import io
            df = pd.read_json(io.StringIO(data_bytes))
            # Re-instantiate adapter in subprocess
            adapter = algo_cls(params=algo_params)
            adj, meta, _ = adapter.fit(df)
            # Only send picklable data back
            conn.send(("ok", adj.tolist(), {k: v for k, v in meta.items()
                        if isinstance(v, (str, int, float, bool, list, dict, type(None)))}))
        except Exception as e:
            conn.send(("error", str(e), type(e).__name__))
        finally:
            conn.close()

    parent_conn, child_conn = mp.Pipe()
    # Serialize data as JSON for cross-process transfer
    data_json = data.to_json()

    p = mp.Process(
        target=_worker,
        args=(child_conn, type(algo), algo.get_params(), data_json, list(data.columns)),
    )
    p.start()
    p.join(timeout=timeout)

    if p.is_alive():
        p.kill()
        p.join()
        raise TimeoutError(f"Algorithm timed out after {timeout}s")

    if parent_conn.poll():
        result = parent_conn.recv()
        if result[0] == "ok":
            import numpy as np
            return np.array(result[1]), result[2]
        else:
            raise RuntimeError(f"Algorithm failed in subprocess: {result[1]}")
    else:
        raise RuntimeError("Algorithm subprocess ended without returning result")
```

```python
# causal_copilot/algorithms/_backends/granger.py
"""Granger Causality backend using statsmodels."""
import numpy as np
import pandas as pd
from typing import Any, Dict, Tuple
from ._base import Backend


class GrangerCausalityBackend(Backend):
    def fit(self, data: pd.DataFrame) -> Tuple[np.ndarray, Dict[str, Any], Any]:
        if 'domain_index' in data.columns:
            data = data.drop(columns=['domain_index'])

        from statsmodels.tsa.stattools import grangercausalitytests

        node_names = list(data.columns)
        n_vars = len(node_names)
        max_lag = self._params.get('p', 10)
        alpha = self._params.get('alpha', 0.05)
        criterion = self._params.get('criterion', 'ssr_ftest')

        adj_matrix = np.zeros((n_vars, n_vars), dtype=int)

        for i in range(n_vars):
            for j in range(n_vars):
                if i == j:
                    continue  # No self-loops
                try:
                    test_result = grangercausalitytests(
                        data[[node_names[i], node_names[j]]].values,
                        maxlag=max_lag,
                        verbose=False,
                    )
                    p_values = [test_result[lag + 1][0][criterion][1]
                                for lag in range(max_lag)]
                    if min(p_values) < alpha:
                        adj_matrix[i, j] = 1  # j Granger-causes i
                except Exception:
                    pass  # Some pairs may fail (e.g. constant columns)

        info = {'lag': max_lag, 'nodes': node_names}
        return adj_matrix, info, None
```
