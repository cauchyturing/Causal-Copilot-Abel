```python
    def test_core_does_not_import_agent(self):
        """causal_copilot/ must never import from agent/."""
        import ast
        from pathlib import Path

        core_dir = Path(__file__).parent.parent / "causal_copilot"
        for py_file in core_dir.rglob("*.py"):
            source = py_file.read_text()
            try:
                tree = ast.parse(source)
            except SyntaxError:
                continue
            for node in ast.walk(tree):
                if isinstance(node, ast.Import):
                    for alias in node.names:
                        assert not alias.name.startswith("agent"), (
                            f"{py_file.name} imports from agent: {alias.name}"
                        )
                elif isinstance(node, ast.ImportFrom):
                    if node.module and node.module.startswith("agent"):
                        raise AssertionError(
                            f"{py_file.name} imports from agent: {node.module}"
                        )
```

```python
"""Evaluation metrics for causal discovery benchmarks.

Implements standard metrics from the causal discovery literature:
- Edge-level: precision, recall, F1, FDR, SHD
- Skeleton-level: ignoring edge direction
- Orientation: accuracy of directed edges
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np


@dataclass
class DiscoveryMetrics:
    """Metrics comparing discovered adjacency to ground truth."""

    # Edge-level (directed)
    true_positives: int
    false_positives: int
    false_negatives: int
    precision: float
    recall: float
    f1: float
    fdr: float  # false discovery rate = FP / (FP + TP)
    shd: int  # Structural Hamming Distance

    # Skeleton-level (undirected)
    skeleton_tp: int
    skeleton_fp: int
    skeleton_fn: int
    skeleton_precision: float
    skeleton_recall: float
    skeleton_f1: float

    # Orientation
    orientation_accuracy: float  # fraction of skeleton-correct edges with correct direction

    def to_dict(self) -> dict:
        return {
            "true_positives": self.true_positives,
            "false_positives": self.false_positives,
            "false_negatives": self.false_negatives,
            "precision": round(self.precision, 4),
            "recall": round(self.recall, 4),
            "f1": round(self.f1, 4),
            "fdr": round(self.fdr, 4),
            "shd": self.shd,
            "skeleton_tp": self.skeleton_tp,
            "skeleton_fp": self.skeleton_fp,
            "skeleton_fn": self.skeleton_fn,
            "skeleton_precision": round(self.skeleton_precision, 4),
            "skeleton_recall": round(self.skeleton_recall, 4),
            "skeleton_f1": round(self.skeleton_f1, 4),
            "orientation_accuracy": round(self.orientation_accuracy, 4),
        }


def _skeleton(adj: np.ndarray) -> np.ndarray:
    """Convert directed adjacency to undirected skeleton (upper triangle)."""
    binary = (adj > 0).astype(int)
    sym = np.maximum(binary, binary.T)
    # Return upper triangle only to avoid double-counting
    return np.triu(sym, k=1)


def evaluate_adjacency(predicted: np.ndarray, ground_truth: np.ndarray) -> DiscoveryMetrics:
    """Compare predicted adjacency matrix against ground truth.

    Both matrices use convention: mat[i,j]=1 means j→i.
    Predicted values >0 are treated as edges (handles 1/2/3 encoding).
    """
    pred_binary = (predicted > 0).astype(int)
    gt_binary = (ground_truth > 0).astype(int)

    # Edge-level metrics (directed)
    tp = int(np.sum((pred_binary == 1) & (gt_binary == 1)))
    fp = int(np.sum((pred_binary == 1) & (gt_binary == 0)))
    fn = int(np.sum((pred_binary == 0) & (gt_binary == 1)))

    precision = tp / (tp + fp) if (tp + fp) > 0 else 0.0
    recall = tp / (tp + fn) if (tp + fn) > 0 else 0.0
    f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0.0
    fdr = fp / (fp + tp) if (fp + tp) > 0 else 0.0

    # SHD = |edges in pred but not gt| + |edges in gt but not pred|
    shd = int(np.sum(pred_binary != gt_binary))

    # Skeleton metrics (ignoring direction)
    skel_pred = _skeleton(pred_binary)
    skel_gt = _skeleton(gt_binary)
    skel_tp = int(np.sum((skel_pred == 1) & (skel_gt == 1)))
    skel_fp = int(np.sum((skel_pred == 1) & (skel_gt == 0)))
    skel_fn = int(np.sum((skel_pred == 0) & (skel_gt == 1)))

    skel_prec = skel_tp / (skel_tp + skel_fp) if (skel_tp + skel_fp) > 0 else 0.0
    skel_rec = skel_tp / (skel_tp + skel_fn) if (skel_tp + skel_fn) > 0 else 0.0
    skel_f1 = 2 * skel_prec * skel_rec / (skel_prec + skel_rec) if (skel_prec + skel_rec) > 0 else 0.0

    # Orientation accuracy: of edges correct in skeleton, how many have correct direction?
    # An edge (i,j) is "skeleton-correct" if both pred and gt have it (in either direction).
    # It has "correct orientation" if pred[i,j]==gt[i,j] AND pred[j,i]==gt[j,i].
    orient_correct = 0
    orient_total = 0
    n = pred_binary.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            pred_has = pred_binary[i, j] or pred_binary[j, i]
            gt_has = gt_binary[i, j] or gt_binary[j, i]
            if pred_has and gt_has:
                orient_total += 1
                if pred_binary[i, j] == gt_binary[i, j] and pred_binary[j, i] == gt_binary[j, i]:
                    orient_correct += 1
    orient_acc = orient_correct / orient_total if orient_total > 0 else 0.0

    return DiscoveryMetrics(
        true_positives=tp,
        false_positives=fp,
        false_negatives=fn,
        precision=precision,
        recall=recall,
        f1=f1,
        fdr=fdr,
        shd=shd,
        skeleton_tp=skel_tp,
        skeleton_fp=skel_fp,
        skeleton_fn=skel_fn,
        skeleton_precision=skel_prec,
        skeleton_recall=skel_rec,
        skeleton_f1=skel_f1,
        orientation_accuracy=orient_acc,
    )
```

```python
"""Standard benchmark datasets for causal discovery evaluation.

Provides well-known Bayesian network benchmark datasets with known ground truth.
Data is generated from the true DAG with fixed seeds for reproducibility.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
import pandas as pd


@dataclass
class StandardDataset:
    name: str
    description: str
    data: pd.DataFrame
    ground_truth: np.ndarray
    columns: list[str]
    source: str  # Citation or reference


def _generate_from_dag(
    columns: list[str],
    edges: list[tuple[int, int, float]],  # (src_idx, tgt_idx, weight)
    n: int = 1000,
    seed: int = 42,
    noise_scale: float = 1.0,
) -> tuple[pd.DataFrame, np.ndarray]:
    """Generate linear Gaussian data from a DAG specification."""
    rng = np.random.default_rng(seed)
    p = len(columns)
    gt = np.zeros((p, p), dtype=int)
    weights = np.zeros((p, p))

    for src, tgt, w in edges:
        gt[tgt, src] = 1  # mat[i,j]=1 means j→i
        weights[tgt, src] = w

    # Topological order: generate data respecting the DAG
    # Simple approach: iterate columns, each depends on parents via weights
    data = np.zeros((n, p))
    for j in range(p):
        parents = np.where(gt[j, :] > 0)[0]
        signal = data[:, parents] @ weights[j, parents] if len(parents) > 0 else 0
        data[:, j] = signal + rng.normal(scale=noise_scale, size=n)

    return pd.DataFrame(data, columns=columns), gt


def _sachs() -> StandardDataset:
    """Sachs et al. (2005) protein signaling network.

    11 proteins, 17 edges. Classic benchmark from flow cytometry data.
    We generate synthetic data matching the known DAG structure.
    """
    columns = ["Raf", "Mek", "PLCg", "PIP2", "PIP3", "Erk", "Akt", "PKA", "PKC", "P38", "JNK"]
    # Ground truth edges from Sachs et al. (2005)
    edges = [
        (2, 3, 0.7),   # PLCg → PIP2
        (2, 4, 0.6),   # PLCg → PIP3
        (4, 3, 0.5),   # PIP3 → PIP2
        (8, 0, 0.6),   # PKC → Raf
        (7, 0, 0.5),   # PKA → Raf
        (7, 1, 0.7),   # PKA → Mek
        (7, 5, 0.6),   # PKA → Erk
        (7, 6, 0.5),   # PKA → Akt
        (7, 10, 0.4),  # PKA → JNK
        (7, 9, 0.5),   # PKA → P38
        (8, 1, 0.5),   # PKC → Mek
        (8, 10, 0.5),  # PKC → JNK
        (8, 9, 0.5),   # PKC → P38
        (0, 1, 0.8),   # Raf → Mek
        (1, 5, 0.8),   # Mek → Erk
        (5, 6, 0.6),   # Erk → Akt
        (4, 6, -0.3),  # PIP3 → Akt
    ]
    data, gt = _generate_from_dag(columns, edges, n=853, seed=42)
    return StandardDataset(
        name="sachs",
        description="Sachs protein signaling (11 nodes, 17 edges)",
        data=data,
        ground_truth=gt,
        columns=columns,
        source="Sachs et al. (2005). Causal protein-signaling networks. Science 308(5721).",
    )


def _asia() -> StandardDataset:
    """Asia (Lauritzen & Spiegelhalter, 1988) lung cancer network.

    8 nodes, 8 edges. Classic textbook Bayesian network.
    """
    columns = ["Asia", "Smoking", "Tuberculosis", "LungCancer", "Bronchitis",
               "TBorCancer", "XRay", "Dyspnea"]
    edges = [
        (0, 2, 0.7),   # Asia → Tuberculosis
        (1, 3, 0.8),   # Smoking → LungCancer
        (1, 4, 0.7),   # Smoking → Bronchitis
        (2, 5, 0.9),   # Tuberculosis → TBorCancer
        (3, 5, 0.9),   # LungCancer → TBorCancer
        (5, 6, 0.8),   # TBorCancer → XRay
        (5, 7, 0.7),   # TBorCancer → Dyspnea
        (4, 7, 0.6),   # Bronchitis → Dyspnea
    ]
    data, gt = _generate_from_dag(columns, edges, n=1000, seed=42)
    return StandardDataset(
        name="asia",
        description="Asia lung cancer network (8 nodes, 8 edges)",
        data=data,
        ground_truth=gt,
        columns=columns,
        source="Lauritzen & Spiegelhalter (1988). Local computations with probabilities. JRSS-B.",
    )


def _alarm() -> StandardDataset:
    """ALARM (Beinlich et al., 1989) medical monitoring network.

    37 nodes, 46 edges. Large benchmark for scalability testing.
    Simplified to continuous linear Gaussian for algorithm compatibility.
    """
    columns = [f"V{i}" for i in range(37)]
    rng = np.random.default_rng(0)
    # Generate a random sparse DAG with 46 edges
    edges = []
    edge_set = set()
    while len(edges) < 46:
        src = rng.integers(0, 30)  # bias toward earlier nodes as sources
        tgt = rng.integers(src + 1, 37)  # ensure DAG (tgt > src)
        if (src, tgt) not in edge_set:
            edge_set.add((src, tgt))
            w = rng.uniform(0.3, 0.8) * rng.choice([-1, 1])
            edges.append((src, tgt, w))

    data, gt = _generate_from_dag(columns, edges, n=2000, seed=42, noise_scale=0.8)
    return StandardDataset(
        name="alarm",
        description="ALARM-like network (37 nodes, 46 edges)",
        data=data,
        ground_truth=gt,
        columns=columns,
        source="Beinlich et al. (1989). The ALARM monitoring system. AIME.",
    )


STANDARD_DATASETS: dict[str, StandardDataset] = {
    "sachs": _sachs(),
    "asia": _asia(),
    "alarm": _alarm(),
}
```

```python
"""Benchmark runner — runs algorithms against scenarios and datasets.

Produces BenchmarkResult with full metrics and provenance.
"""

from __future__ import annotations

import time
from dataclasses import dataclass
from typing import Any

import numpy as np
import pandas as pd

from benchmarks.evaluate import DiscoveryMetrics, evaluate_adjacency
from benchmarks.scenarios import ALL_SCENARIOS
from benchmarks.datasets import STANDARD_DATASETS


@dataclass
class BenchmarkResult:
    algorithm: str
    scenario: str
    metrics: DiscoveryMetrics
    adjacency_matrix: np.ndarray
    provenance: Any  # Provenance from CausalResult
    runtime_seconds: float

    def to_dict(self) -> dict:
        return {
            "algorithm": self.algorithm,
            "scenario": self.scenario,
            "metrics": self.metrics.to_dict(),
            "runtime_seconds": round(self.runtime_seconds, 3),
            "provenance": {
                "seed": self.provenance.seed if self.provenance else None,
                "algorithm_version": self.provenance.algorithm_version if self.provenance else None,
            },
        }


def run_benchmark(
    algorithm: str,
    scenario_name: str,
    seed: int = 42,
    timeout: int = 120,
) -> BenchmarkResult:
    """Run a single algorithm on a single scenario/dataset and evaluate."""
    from causal_copilot import CausalCopilot

    # Look up scenario or dataset
    if scenario_name in ALL_SCENARIOS:
        scenario = ALL_SCENARIOS[scenario_name]
        data = scenario.data
        ground_truth = scenario.ground_truth
    elif scenario_name in STANDARD_DATASETS:
        ds = STANDARD_DATASETS[scenario_name]
        data = ds.data
        ground_truth = ds.ground_truth
    else:
        raise ValueError(f"Unknown scenario/dataset: {scenario_name}")

    start = time.monotonic()
    copilot = CausalCopilot()
    result = copilot.analyze(data, algorithm=algorithm, seed=seed, timeout=timeout)
    elapsed = time.monotonic() - start

    if result.status != "ok" or result.adjacency_matrix is None:
        # Return zero metrics for failed runs
        n = ground_truth.shape[0]
        return BenchmarkResult(
            algorithm=algorithm,
            scenario=scenario_name,
            metrics=evaluate_adjacency(np.zeros_like(ground_truth), ground_truth),
            adjacency_matrix=np.zeros_like(ground_truth),
            provenance=result.provenance,
            runtime_seconds=elapsed,
        )

    # Trim ground truth if matrix sizes differ (e.g. domain_index dropped)
    adj = result.adjacency_matrix
    if adj.shape != ground_truth.shape:
        gt = ground_truth[: adj.shape[0], : adj.shape[1]]
    else:
        gt = ground_truth

    metrics = evaluate_adjacency(adj, gt)

    return BenchmarkResult(
        algorithm=algorithm,
        scenario=scenario_name,
        metrics=metrics,
        adjacency_matrix=adj,
        provenance=result.provenance,
        runtime_seconds=elapsed,
    )
```

```python
# causal_copilot/agent/llm.py
"""Minimal LLM client — OpenAI-compatible API for all providers.

Uses the OpenAI Python SDK which speaks to any OpenAI-compatible endpoint:
openai.com, OpenRouter, Ollama (/v1), LM Studio (/v1).

One class, one file, no Pydantic dependency.
"""

from __future__ import annotations

import json
import os

PROVIDERS: dict[str, dict] = {
    "openai": {
        "base_url": "https://api.openai.com/v1",
        "default_model": "gpt-4o-mini",
        "key_env": "OPENAI_API_KEY",
    },
    "openrouter": {
        "base_url": "https://openrouter.ai/api/v1",
        "default_model": "openai/gpt-4o-mini",
        "key_env": "OPENROUTER_API_KEY",
    },
    "ollama": {
        "base_url": "http://localhost:11434/v1",
        "default_model": "llama3.2",
        "key_env": None,
    },
    "lmstudio": {
        "base_url": "http://localhost:1234/v1",
        "default_model": "default",
        "key_env": None,
    },
}


class AgentLLM:
    """Thin OpenAI-compatible LLM client with provider presets.

    All providers use the same OpenAI SDK — Ollama and LM Studio expose
    an OpenAI-compatible /v1 endpoint, so no special client needed.
    """

    def __init__(
        self,
        provider: str = "openai",
        model: str | None = None,
        api_key: str | None = None,
        base_url: str | None = None,
    ):
        preset = PROVIDERS.get(provider, PROVIDERS["openai"])
        self.provider = provider
        self.model = model or os.getenv("LLM_MODEL") or preset["default_model"]
        self._base_url = base_url or os.getenv("LLM_BASE_URL") or preset["base_url"]

        key_env = preset.get("key_env")
        resolved_key = api_key or (os.getenv(key_env) if key_env else None) or "not-needed"

        # Lazy import — openai is an optional dep ([agent] extra)
        try:
            from openai import OpenAI
        except ImportError:
            raise ImportError(
                "Agent mode requires openai. Install with: pip install causal-copilot[agent]"
            )
        self._client = OpenAI(api_key=resolved_key, base_url=self._base_url)

    # Providers that support response_format={"type":"json_object"}
    _JSON_MODE_PROVIDERS = {"openai", "openrouter", "ollama"}

    def complete(
        self,
        prompt: str,
        system: str = "You are a causal inference expert.",
        json_mode: bool = False,
    ) -> str | dict:
        """Single-turn completion. Returns str or parsed dict (if json_mode).

        P1 fix: JSON mode is provider-aware. LM Studio doesn't support
        response_format, so we fall back to prompt-based JSON extraction.
        """
        messages = [
            {"role": "system", "content": system},
            {"role": "user", "content": prompt},
        ]
        kwargs: dict = {
            "model": self.model,
            "messages": messages,
            "temperature": 0.0,
        }

        use_json_format = json_mode and self.provider in self._JSON_MODE_PROVIDERS
        if use_json_format:
            kwargs["response_format"] = {"type": "json_object"}
            if "json" not in system.lower() and "json" not in prompt.lower():
                messages[0]["content"] += " Respond in JSON format."
        elif json_mode:
            # Provider doesn't support response_format — ask via prompt
            messages[0]["content"] += " You MUST respond with valid JSON only."

        response = self._client.chat.completions.create(**kwargs)
        content = response.choices[0].message.content

        if json_mode:
            try:
                return json.loads(content)
            except json.JSONDecodeError:
                # Try to extract JSON from markdown code block
                import re
                match = re.search(r"```json\s*(.*?)\s*```", content, re.DOTALL)
                if match:
                    return json.loads(match.group(1))
                raise
        return content
```

```python
# causal_copilot/agent/selector.py
"""LLM-driven algorithm selection and hyperparameter tuning.

Reads algorithm profiles and HP specs from packaged context files.
Only selects from algorithms registered in the clean core REGISTRY.
"""

from __future__ import annotations

import json
from pathlib import Path

from causal_copilot.agent import _CONTEXT_DIR
from causal_copilot.agent.llm import AgentLLM
from causal_copilot.algorithms.registry import REGISTRY


def _load_text(rel_path: str) -> str:
    return (_CONTEXT_DIR / rel_path).read_text(encoding="utf-8")


def _load_json(rel_path: str) -> dict:
    return json.loads((_CONTEXT_DIR / rel_path).read_text(encoding="utf-8"))


def _build_algorithm_catalog() -> dict[str, dict]:
    """Build catalog of registered algorithms with context profiles and HP specs.

    Only includes algorithms that are BOTH in the REGISTRY and have context files.
    """
    catalog: dict[str, dict] = {}
    for name in REGISTRY:
        profile_path = _CONTEXT_DIR / "algos" / f"{name}.txt"
        hp_path = _CONTEXT_DIR / "hyperparameters" / f"{name}.json"
        if profile_path.exists():
            catalog[name] = {
                "profile": profile_path.read_text(encoding="utf-8"),
                "hyperparameters": (
                    json.loads(hp_path.read_text(encoding="utf-8"))
                    if hp_path.exists()
                    else {}
                ),
            }
    return catalog


@dataclass
class AgentDecision:
    """Output of LLM-driven algorithm selection."""
    algorithm: str
    hyperparams: dict
    reasoning: str
    source: str  # "llm" or "rule_fallback"


def select_algorithm(
    llm: AgentLLM,
    data_properties: dict,
    query: str = "",
) -> AgentDecision:
    """Use LLM to select the best algorithm for the given data.

    P0 fix: wraps LLM call in try/except — falls back to rule_based_select()
    on ANY failure (auth, network, JSON parse, unknown algo).
    P1 fix: returns AgentDecision with reasoning, not just algorithm name.

    Returns:
        AgentDecision (algorithm guaranteed to be in REGISTRY).
    """
    from causal_copilot.core.planner import rule_based_select

    catalog = _build_algorithm_catalog()
    available = sorted(catalog.keys())

    # Build algorithm summaries for the prompt
    algo_summaries = ""
    for name in available:
        profile = catalog[name]["profile"][:600]
        algo_summaries += f"\n### {name}\n{profile}\n"

    prompt = f"""You are selecting a causal discovery algorithm.

## Data Properties
- Samples: {data_properties['n_samples']}
- Features: {data_properties['n_features']}
- Likely linear: {data_properties['likely_linear']}
- Likely Gaussian: {data_properties['likely_gaussian']}
- Time series: {data_properties['is_time_series']}

## User Query
{query or 'Discover causal relationships in this dataset.'}

## Available Algorithms
{algo_summaries}

Select the SINGLE best algorithm for this data. You MUST choose from: {available}

Respond in JSON: {{"algorithm": "<name>", "reasoning": "<1-2 sentences>"}}"""

    try:
        result = llm.complete(prompt, json_mode=True)
        selected = result.get("algorithm", "")
        reasoning = result.get("reasoning", "")

        if selected in REGISTRY:
            return AgentDecision(
                algorithm=selected,
                hyperparams={},
                reasoning=reasoning,
                source="llm",
            )
    except Exception:
        pass  # Fall through to rule-based fallback

    # Fallback: rule-based selection (always works, no LLM needed)
    decision = rule_based_select(data_properties)
    return AgentDecision(
        algorithm=decision.algorithm,
        hyperparams=decision.hyperparams,
        reasoning=f"[LLM fallback] {decision.reason}",
        source="rule_fallback",
    )


def tune_hyperparameters(
    llm: AgentLLM,
    algorithm: str,
    data_properties: dict,
) -> dict:
    """Use LLM to tune hyperparameters for the selected algorithm.

    Args:
        llm: AgentLLM instance.
        algorithm: Selected algorithm name.
        data_properties: Output of detect_data_properties().

    Returns:
        Dict of hyperparameter name -> value.
    """
    catalog = _build_algorithm_catalog()
    info = catalog.get(algorithm)
    if info is None:
        return {}

    hp_spec = info["hyperparameters"]
    if not hp_spec:
        return {}

    # Convert HP spec to readable format
    hp_description = ""
    for param, details in hp_spec.items():
        if param == "algorithm_name":
            continue
        if isinstance(details, dict) and "meaning" in details:
            hp_description += f"\n**{param}**: {details['meaning']}\n"
            if "available_values" in details:
                hp_description += f"  Values: {details['available_values']}\n"
            if "expert_suggestion" in details:
                hp_description += f"  Expert suggestion: {details['expert_suggestion']}\n"

    prompt = f"""You are tuning hyperparameters for the {algorithm} causal discovery algorithm.

## Data Properties
- Samples: {data_properties['n_samples']}
- Features: {data_properties['n_features']}
- Likely linear: {data_properties['likely_linear']}
- Likely Gaussian: {data_properties['likely_gaussian']}

## Available Hyperparameters
{hp_description}

Select optimal values. Respond in JSON with parameter names as keys and values.
Only include parameters you want to change from defaults."""

    try:
        result = llm.complete(prompt, json_mode=True)
    except Exception:
        return {}  # LLM failure -> use defaults

    # Filter to only valid parameter names
    valid_params = {
        k: v for k, v in result.items()
        if k in hp_spec and k != "algorithm_name"
    }
    return valid_params
```

```python
# causal_copilot/agent/pipeline.py
"""AgentCopilot — LLM-driven autonomous causal analysis.

Orchestrates the full agent pipeline:
1. Detect data properties (reuses core/planner.py)
2. LLM selects algorithm (from registered algorithms with context profiles)
3. LLM tunes hyperparameters (from HP specs)
4. Execute algorithm (delegates to CausalCopilot — subprocess isolated)
5. LLM interprets result (optional natural language summary)

The agent ADDS intelligence on top of the core — it never bypasses it.
"""

from __future__ import annotations

from pathlib import Path

import pandas as pd

from causal_copilot import CausalCopilot
from causal_copilot.core.planner import detect_data_properties
from causal_copilot.core.result import CausalResult


class AgentCopilot:
    """LLM-driven causal analysis agent.

    Usage:
        agent = AgentCopilot(provider="openai")
        result = agent.analyze(df, query="What causes Y?")

    Provider presets: "openai", "openrouter", "ollama", "lmstudio"
    Or pass base_url for any OpenAI-compatible endpoint.
    """

    def __init__(
        self,
        provider: str = "openai",
        model: str | None = None,
        api_key: str | None = None,
        base_url: str | None = None,
    ):
        from causal_copilot.agent.llm import AgentLLM

        self.llm = AgentLLM(
            provider=provider, model=model,
            api_key=api_key, base_url=base_url,
        )
        self._copilot = CausalCopilot(planner="rule")

    def analyze(
        self,
        data: str | Path | pd.DataFrame,
        *,
        query: str = "",
        timeout: int = 300,
        seed: int = 42,
    ) -> CausalResult:
        """Run LLM-driven causal analysis.

        Args:
            data: CSV file path or DataFrame.
            query: User's causal question (helps LLM select algorithm).
            timeout: Max seconds for algorithm execution.
            seed: Random seed for reproducibility.

        Returns:
            CausalResult with graph, provenance (planner="llm"), and summary.
        """
        from causal_copilot.agent.selector import (
            select_algorithm,
            tune_hyperparameters,
        )
        import numpy as np

        # Load data if path
        if isinstance(data, (str, Path)):
            df = pd.read_csv(data)
        else:
            df = data

        # Detect data properties (reuse core logic)
        numeric_df = df.select_dtypes(include=[np.number])
        props = detect_data_properties(numeric_df)

        # Step 1: LLM selects algorithm (P0: returns AgentDecision, handles errors)
        agent_decision = select_algorithm(self.llm, props, query=query)

        # Step 2: LLM tunes hyperparameters (merges into agent_decision.hyperparams)
        tuned_hp = tune_hyperparameters(self.llm, agent_decision.algorithm, props)
        effective_hp = {**agent_decision.hyperparams, **tuned_hp}

        # Step 3: Execute via core — P0 fix: pass algorithm_params so HPs reach execution
        result = self._copilot.analyze(
            data=df,
            algorithm=agent_decision.algorithm,
            algorithm_params=effective_hp,  # P0: LLM-tuned HPs reach adapter
            timeout=timeout,
            seed=seed,
        )

        # P1: Thread LLM reasoning into provenance + algorithm_selection_reason
        if result.provenance is not None:
            from dataclasses import replace
            result.provenance = replace(
                result.provenance,
                planner="llm" if agent_decision.source == "llm" else "rule",
                planner_model=self.llm.model,
            )
        result.algorithm_selection_reason = agent_decision.reasoning

        # Step 4: LLM interprets result (P2: stored in discovery_metadata, not summary)
        self._interpret_result(result, query)

        return result

    def _interpret_result(self, result: CausalResult, query: str) -> None:
        """Add LLM-generated interpretation to result (stored in discovery_metadata)."""
        if result.status != "ok" or result.adjacency_matrix is None:
            return

        # Build edge list for LLM
        edges = []
        names = result.node_names or []
        adj = result.adjacency_matrix
        for i in range(adj.shape[0]):
            for j in range(adj.shape[1]):
                if adj[i, j] == 1 and i < len(names) and j < len(names):
                    edges.append(f"{names[j]} -> {names[i]}")
                elif adj[i, j] == 2 and i < len(names) and j < len(names):
                    edges.append(f"{names[j]} -- {names[i]}")

        edge_text = "\n".join(edges) if edges else "No edges discovered."

        prompt = f"""A causal discovery algorithm found these relationships:
{edge_text}

User question: {query or 'What are the causal relationships?'}

Provide a brief (2-3 sentence) scientific interpretation."""

        try:
            interpretation = self.llm.complete(prompt)
            # P2: Store separately, don't mutate core summary
            result.discovery_metadata["agent_interpretation"] = interpretation
        except Exception:
            pass  # Keep result clean if LLM fails
```
